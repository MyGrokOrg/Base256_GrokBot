cmake_minimum_required(VERSION 3.20)
project(Base256 VERSION 4.14.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(base256 INTERFACE)
add_library(Base256::base256 ALIAS base256)
target_include_directories(base256 INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_float256 tests/test_float256.cpp)
    target_link_libraries(test_float256 PRIVATE base256)
    add_executable(test_extra_math tests/test_extra_math.cpp)
    target_link_libraries(test_extra_math PRIVATE base256)
    if(MSVC)
        target_compile_options(test_float256 PRIVATE /W4 /permissive-)
        target_compile_options(test_extra_math PRIVATE /W4 /permissive-)
    else()
        target_compile_options(test_float256 PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(test_extra_math PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME float256 COMMAND test_float256)
    add_test(NAME extra_math COMMAND test_extra_math)
    add_executable(test_fenv tests/test_fenv.cpp)
    target_link_libraries(test_fenv PRIVATE base256)
    add_test(NAME fenv COMMAND test_fenv)
    add_executable(test_math_flags tests/test_math_flags.cpp)
    target_link_libraries(test_math_flags PRIVATE base256)
    add_test(NAME math_flags COMMAND test_math_flags)
    # Fast paths of + - * must match the general path bit for bit (flags included).
    add_executable(fastpath_fast tests/fastpath_diff.cpp)
    target_link_libraries(fastpath_fast PRIVATE base256)
    add_executable(fastpath_slow tests/fastpath_diff.cpp)
    target_link_libraries(fastpath_slow PRIVATE base256)
    target_compile_definitions(fastpath_slow PRIVATE BASE256_NO_FAST_PATHS)
    set(BASE256_FASTPATH_CASES "200000" CACHE STRING "Random cases for the fastpath test")
    add_test(NAME fastpath COMMAND ${CMAKE_COMMAND} -DA=$<TARGET_FILE:fastpath_fast> -DB=$<TARGET_FILE:fastpath_slow>
             -DN=${BASE256_FASTPATH_CASES} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/fastpath_compare.cmake)
    # Benchmark (not a test): build/base256_bench [min_seconds] [filter]
    add_executable(base256_bench bench/bench.cpp)
    target_link_libraries(base256_bench PRIVATE base256)
    add_executable(test_format tests/test_format.cpp)
    target_link_libraries(test_format PRIVATE base256)
    add_test(NAME format COMMAND test_format)
    add_executable(test_macros tests/test_macros.cpp)
    target_link_libraries(test_macros PRIVATE base256)
    add_test(NAME macros COMMAND test_macros)

    # Differential testing against GNU MPFR (237-bit, binary256 exponent range,
    # subnormalized) with randomized operands. Requires libmpfr + libgmp.
    option(BASE256_MPFR_TESTS "Build the MPFR differential/fuzz test" OFF)
    set(BASE256_MPFR_CASES "5000" CACHE STRING "Random cases per op for the ctest MPFR run")
    if(BASE256_MPFR_TESTS)
        find_path(MPFR_INCLUDE_DIR mpfr.h REQUIRED)
        find_library(MPFR_LIBRARY mpfr REQUIRED)
        find_library(GMP_LIBRARY gmp REQUIRED)
        add_executable(mpfr_diff tests/mpfr_diff.cpp)
        target_include_directories(mpfr_diff PRIVATE ${MPFR_INCLUDE_DIR})
        target_link_libraries(mpfr_diff PRIVATE base256 ${MPFR_LIBRARY} ${GMP_LIBRARY})
        if(NOT MSVC)
            target_compile_options(mpfr_diff PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        add_test(NAME mpfr_diff COMMAND mpfr_diff ${BASE256_MPFR_CASES} 12345)
        # Elementary-function error bounds (ulps) against a 1100-bit MPFR reference.
        set(BASE256_ULP_CASES "200" CACHE STRING "Random cases per function for the ctest ulp run")
        set(BASE256_ULP_FUNCS "exp exp2 expm1 exp_big exp2_big expm1_big sinh_big cosh_big log log2 log10 logn1 log1p log1pn cbrt pow sin cos tan sin_kpi cos_kpi tan_kpi sin_all cos_all tan_all sin_kbig sin_hard cos_hard tan_hard sin_hardcw atan asin acos asin1 acos1 atan2 sinh cosh tanh asinh acosh atanh atanh1 acosh1 erf erfc erfcn erf_lin erfc_lin tgamma tgamman lgamma lgamman lgamma1 lgamma2")
        add_executable(mpfr_ulp tests/mpfr_ulp.cpp)
        target_include_directories(mpfr_ulp PRIVATE ${MPFR_INCLUDE_DIR})
        target_link_libraries(mpfr_ulp PRIVATE base256 ${MPFR_LIBRARY} ${GMP_LIBRARY})
        if(NOT MSVC)
            target_compile_options(mpfr_ulp PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        add_test(NAME mpfr_ulp COMMAND mpfr_ulp ${BASE256_ULP_CASES} 777 1 "${BASE256_ULP_FUNCS}")
    endif()
endif()

add_executable(example_basic examples/basic.cpp)
target_link_libraries(example_basic PRIVATE base256)

add_executable(example_high_precision examples/high_precision.cpp)
target_link_libraries(example_high_precision PRIVATE base256)

install(DIRECTORY include/base256 DESTINATION include)
